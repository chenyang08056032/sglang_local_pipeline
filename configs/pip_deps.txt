# 单机共享容器初始化时执行的 python 依赖安装命令 (固定默认文件, 零配置:
# 存在即生效, 不想装任何依赖可清空或删除本文件)
# 格式: 每行一条命令按序执行 (同一 shell, cd/变量状态延续); # 注释与空行忽略
# 内容对齐 CI .github/workflows/_npu-pr-test-stage.yml 的 Install dependencies 步骤,
# CI 更新后直接把对应行抄过来即可; 单条失败只记 WARN 跳过继续 (不终止初始化,
# 看 runs/{run_id}/.init.log 里的 [pip_deps] [WARN] 行); 注意不要写 exit, 会终止整个初始化
# 每节点只装一次, 该节点全部单机用例复用; 多机用例容器不执行本文件 (依赖需打进镜像)

# 对齐 CI 的执行目录 (repo 根); /root/sglang 是初始化建好的 repo 软链
cd /root/sglang

# ---- 基础依赖 ----
pip install sentence_transformers zss "wandb>=0.16.0" tenacity==8.3.0 loguru openpyxl latex2sympy2 zstandard transformers-stream-generator tqdm-multiprocess pycocoevalcap
pip install yt-dlp sentencepiece==0.1.99 nltk av ftfy sqlitedict==2.1.0 "sacrebleu>=1.5.0" pytablewriter "peft>=0.2.0" "accelerate>=0.29.1"
pip install jsonlines "evaluate>=0.4.0" numexpr dotenv
pip install websocket-client
pip install watchfiles granian whisper_normalizer msgpack
pip install fastokens
pip install mindstudio-probe --pre
pip install "antlr4-python3-runtime==4.9.3" --force-reinstall --no-deps
pip install "uvicorn[standard]"

# ---- sgl-eval (source CI 脚本取安装规格后安装) ----
sglang_source_path=$(pwd)
echo "Source code path: ${sglang_source_path}"
. "${sglang_source_path}/scripts/ci/utils/sgl_eval_ref.sh"
pip install "$SGL_EVAL_SPEC"

# ---- lmms-eval (源码安装 + 去掉 decord 依赖) ----
# clone 到 /tmp (容器内 = runs/{run_id}/tmp, 随 run 自动清理): 不能 clone 到当前目录,
# /root/sglang 是执行机持久仓的软链, 落进去会污染仓且二次 run 时 clone 目标已存在必失败;
# 全程绝对路径不 cd: clone 失败时后续行各自失败被逐条跳过, 不会误在别的目录执行
git clone --branch v0.3.3 --depth 1 https://github.com/EvolvingLMMs-Lab/lmms-eval.git /tmp/lmms-eval
sed -i "/decord; platform_system/d" /tmp/lmms-eval/pyproject.toml
python3 -m pip install /tmp/lmms-eval > /tmp/lmms-eval/lmmslog.txt 2>&1 || { cat /tmp/lmms-eval/lmmslog.txt; false; }
python3 -c "import lmms_eval"
hf download MMMU/MMMU --repo-type dataset
pip install opentelemetry-api opentelemetry-sdk opentelemetry-exporter-otlp opentelemetry-exporter-otlp-proto-grpc opentelemetry-exporter-otlp-proto-http

# ---- sglang_router ----
apt-get install -y libssl-dev
pip install sglang_router

# ---- numactl (流水线自带 aarch64 二进制: 本流水线仓 assets/numactl/) ----
# 镜像缺 numactl/libnuma 时手动装 (apt 源不可用场景); 装到 /usr/local/bin 与
# /usr/local/lib 而非 ~/numactl + export: 用例走独立 docker exec, 不继承初始化
# shell 的 PATH/LD_LIBRARY_PATH, 只有系统默认路径才对后续用例生效
# assets 由流水线启动容器前从本流水线仓分发到节点 tmp 目录 (容器内
# /tmp/sglang_pipeline_assets), 故用 $SGLANG_PIPELINE_ASSETS 引用:
# 流水线是独立仓, 节点上只有 sglang 仓副本, 不能写成 sglang 仓内路径
mkdir -p /usr/local/bin /usr/local/lib
install -m 755 "$SGLANG_PIPELINE_ASSETS/numactl/numactl" /usr/local/bin/
install -m 644 "$SGLANG_PIPELINE_ASSETS/numactl/libnuma.so.1" /usr/local/lib/
ldconfig

# ---- pyarrow/numpy 兼容 ----
# pyarrow>=19 要求 numpy>=2.0, 但 lmms-eval v0.3.3 会把 numpy 钉回 1.26.4,
# 导致 import datasets/pyarrow 报 "pyarrow requires NumPy 2.0 or newer, found 1.26.4"
# 放末尾: 保证后续安装不再把 pyarrow 拉回 >=19, 最终统一降到 18.x (兼容 numpy 1.26)
pip install "pyarrow<19"
